## 测试夹具原则

**核心约束：所有测试使用临时隔离目录，绝不碰用户真实数据。**

```rust
// Rust 测试夹具模式
use tempfile::TempDir;

#[fixture]
fn test_workspace() -> TempDir {
    let dir = TempDir::new().unwrap();
    // 构造测试文件结构
    fs::create_dir_all(dir.path().join("subdir/nested")).unwrap();
    fs::write(dir.path().join("file1.txt"), "content").unwrap();
    fs::write(dir.path().join("subdir/file2.txt"), "data").unwrap();
    dir  // 自动清理
}
```

```typescript
// 前端测试夹具模式
import { mkdtemp } from 'fs/promises';
import { tmpdir } from 'os';

async function createTestWorkspace() {
  const dir = await mkdtemp(path.join(tmpdir(), 'z-file-test-'));
  // 构造测试结构...
  return { dir, cleanup: () => rm(dir, { recursive: true }) };
}
```
